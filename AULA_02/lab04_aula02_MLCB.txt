import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.naive_bayes import MultinomialNB
from sklearn.metrics import classification_report

# 1. Dataset com 15 frases distribuídas em 3 intenções
data = {
    'texto': [
        'Quero comprar uma passagem para Paris',
        'Gostaria de reservar um voo para Nova York',
        'Preciso de uma passagem aérea para Salvador',
        'Como faço para adquirir um bilhete para Tóquio?',
        'Quero agendar uma viagem para Florianópolis',
        'Preciso cancelar minha reserva de voo',
        'Gostaria de solicitar o cancelamento da minha passagem',
        'Como faço para cancelar meu pedido?',
        'Quero cancelar a viagem marcada para a semana que vem',
        'Desejo anular minha reserva por favor',
        'Quero falar com um atendente humano',
        'Preciso de suporte com uma pessoa real',
        'Pode me transferir para um operador?',
        'Gostaria de conversar com a central de atendimento',
        'Preciso de ajuda de um especialista humano'
    ],
    'intencao': [
        'comprar_passagem', 'comprar_passagem', 'comprar_passagem', 'comprar_passagem', 'comprar_passagem',
        'cancelar_reserva', 'cancelar_reserva', 'cancelar_reserva', 'cancelar_reserva', 'cancelar_reserva',
        'falar_atendente', 'falar_atendente', 'falar_atendente', 'falar_atendente', 'falar_atendente'
    ]
}

df = pd.DataFrame(data)

# 2. Divisão em Treino e Teste (com stratify para manter a proporção das intenções)
X_train, X_test, y_train, y_test = train_test_split(
    df['texto'], 
    df['intencao'], 
    test_size=0.2, 
    random_state=42, 
    stratify=df['intencao']
)

# 3. Vetorização de texto utilizando TF-IDF
vectorizer = TfidfVectorizer(ngram_range=(1, 2))
X_train_vec = vectorizer.fit_transform(X_train)
X_test_vec = vectorizer.transform(X_test)

# 4. Escolha e Treinamento do Algoritmo (Multinomial Naive Bayes)
model = MultinomialNB()
model.fit(X_train_vec, y_train)

# 5. Predições com Frases Inéditas
frases_ineditas = [
    'Preciso comprar um bilhete de avião para o Rio de Janeiro',
    'Gostaria de cancelar o meu voo do mês que vem',
    'Por favor me passe para um suporte humano'
]

# Vetorização e Predição das frases inéditas
ineditas_vec = vectorizer.transform(frases_ineditas)
predicoes = model.predict(ineditas_vec)

# 6. Exibição dos resultados das frases inéditas no console
print("=== RESULTADOS DAS FRASES INÉDITAS ===")
for frase, intencao in zip(frases_ineditas, predicoes):
    print(f"Frase: '{frase}' -> Intenção Prevista: [{intencao}]")
